1. SALESMAN1
SQL> CREATE TABLE SALESMAN1(SAL_ID NUMBER(3) PRIMARY KEY,NAME VARCHAR(10),CITY VARCHAR(10),COMMISSION NUMBER(3));

2.CUSTOMER1
SQL> CREATE TABLE CUSTOMER1(CUST_ID NUMBER(4) PRIMARY KEY,CNAME VARCHAR(10),CITY VARCHAR(10),
GRADE NUMBER(5),SAL_ID NUMBER(3),FOREIGN KEY (SAL_ID) REFERENCES SALESMAN1(SAL_ID));

3.ORDER1
SQL> CREATE TABLE ORDER1(ORD_ID NUMBER(3) PRIMARY KEY,PUR_AMT NUMBER(4),ORD_DATE DATE,CUST_ID NUMBER
(4),SAL_ID NUMBER(3),FOREIGN KEY (CUST_ID) REFERENCES CUSTOMER1(CUST_ID),FOREIGN KEY (SAL_ID)REFEREN
CES SALESMAN1(SAL_ID));

4.DEPARTMENT_1
SQL> CREATE TABLE DEPARTMENT_1(DEPT_NO NUMBER(3) PRIMARY KEY,DNAME VARCHAR(10));

5.E_DETAILS
SQL> CREATE TABLE E_DETAILS(EMP_ID NUMBER(5) PRIMARY KEY,E_FNAME VARCHAR2(10),
E_LNAME VARCHAR2(10),SALARY NUMBER(5), DEPT_NO NUMBER(3),
FOREIGN KEY (DEPT_NO) REFERENCES DEPARTMENT_1(DEPT_NO));
