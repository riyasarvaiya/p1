CREATE TABLE Salesman_1 (
    Sid number (5) PRIMARY KEY,
    name VARCHAR(10),
    city VARCHAR(10),
    commission number(5)
);
CREATE TABLE Customer_1 (
    C_id number(5)  PRIMARY KEY,
    cname VARCHAR(10),
    city VARCHAR(10),
    grade number(5),
    salesman_id number(5),
    FOREIGN KEY (salesman_id) REFERENCES Salesman(Sid)
);
